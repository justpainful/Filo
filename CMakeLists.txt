file(MAKE_DIRECTORY ${SHADER_OUT})
function(filo_shader name file entry profile)
  set(out ${SHADER_OUT}/${name}.h)
  add_custom_command(OUTPUT ${out}
    COMMAND ${FXC} /nologo /T ${profile} /E ${entry} /O3 /Zpr /Vn g_${name} /Fh ${out} ${CMAKE_SOURCE_DIR}/${file}
    DEPENDS ${CMAKE_SOURCE_DIR}/${file} ${CMAKE_SOURCE_DIR}/src/gfx/shaders/common.hlsli
    COMMENT "fxc ${name}" VERBATIM)
  set_property(GLOBAL APPEND PROPERTY FILO_SHADER_OUTPUTS ${out})
endfunction()

# ---- modules --------------------------------------------------------------------------------
function(filo_module name)
  file(GLOB_RECURSE srcs CONFIGURE_DEPENDS src/${name}/*.cpp src/${name}/*.h)
  file(GLOB_RECURSE cpps CONFIGURE_DEPENDS src/${name}/*.cpp)
  if(cpps)
    add_library(filo_${name} STATIC ${srcs})
    target_include_directories(filo_${name} PUBLIC ${CMAKE_SOURCE_DIR}/src ${SHADER_OUT})
    add_dependencies(filo_${name} filo_shaders)
  else()
    add_library(filo_${name} INTERFACE)
    target_include_directories(filo_${name} INTERFACE ${CMAKE_SOURCE_DIR}/src ${SHADER_OUT})
  endif()
endfunction()

function(filo_link name)
  get_target_property(_t filo_${name} TYPE)
  if(_t STREQUAL "INTERFACE_LIBRARY")
    target_link_libraries(filo_${name} INTERFACE ${ARGN})
  else()
    target_link_libraries(filo_${name} PUBLIC ${ARGN})
  endif()
endfunction()

set(FILO_EXCLUDE_SRC "" CACHE STRING "Engine source files (basenames) left out of the build while they are being written")
option(FILO_WITH_ENGINES "Link the processing engines into Filo/filo-cli" ON)
set(_excl "")
if(NOT FILO_WITH_ENGINES)
  set(_excl EXCLUDE_FROM_ALL)   # engines are built only when a target asks for them (lets UI work proceed while engines are mid-edit)
endif()
# ---- engines: one static library per src/engines/<name> (independently buildable / testable) -----------------------
file(GLOB _engine_dirs LIST_DIRECTORIES true CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/engines/*)
set(FILO_ENGINE_LIBS "")
foreach(_d ${_engine_dirs})
  if(IS_DIRECTORY ${_d})
    get_filename_component(_n ${_d} NAME)
    file(GLOB_RECURSE _cpps CONFIGURE_DEPENDS ${_d}/*.cpp)
    if(FILO_EXCLUDE_SRC)
      foreach(_x ${FILO_EXCLUDE_SRC})
        list(FILTER _cpps EXCLUDE REGEX ".*/${_x}$")
      endforeach()
    endif()
    if(_cpps)
      add_library(filo_eng_${_n} STATIC ${_excl} ${_cpps})
      target_include_directories(filo_eng_${_n} PUBLIC ${CMAKE_SOURCE_DIR}/src)
      target_link_libraries(filo_eng_${_n} PUBLIC filo_registry filo_core filo_miniz filo_pdfium windowscodecs)
      if(NOT _n STREQUAL "common")
        target_link_libraries(filo_eng_${_n} PUBLIC filo_eng_common)
      endif()
      # optional per-engine CMake hook: cmake/engines/<name>.cmake may add link libraries / definitions to filo_eng_<name>
      if(EXISTS ${CMAKE_SOURCE_DIR}/cmake/engines/${_n}.cmake)
        include(${CMAKE_SOURCE_DIR}/cmake/engines/${_n}.cmake)
      endif()
      list(APPEND FILO_ENGINE_LIBS filo_eng_${_n})
      # per-engine tests: tests/<name>/*.cpp -> filo_tests_<name>
      file(GLOB _tcpps CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/${_n}/*.cpp)
      if(_tcpps)
        add_executable(filo_tests_${_n} ${_excl} ${_tcpps} ${CMAKE_SOURCE_DIR}/tests/test_main.cpp)
        target_include_directories(filo_tests_${_n} PRIVATE ${CMAKE_SOURCE_DIR}/tests)
        target_link_libraries(filo_tests_${_n} PRIVATE filo_eng_${_n} filo_registry filo_core)
        filo_embed_manifest(filo_tests_${_n})
        filo_copy_runtime(filo_tests_${_n})
      endif()
    endif()
  endif()
endforeach()
